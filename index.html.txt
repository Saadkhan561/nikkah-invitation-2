$('#wa').href='https://wa.me/?text='+encodeURIComponent(text+' '+url);
function copy(){if(!url){say('Add your live link in SITE_URL first');return}
  function fb(){try{var a=document.createElement('textarea');a.value=url;a.style.cssText='position:fixed;opacity:0';document.body.appendChild(a);a.select();document.execCommand('copy');a.remove();say('Link copied')}catch(e){say('Copy failed')}}
  if(navigator.clipboard&&navigator.clipboard.writeText)navigator.clipboard.writeText(url).then(function(){say('Link copied')},fb);else fb()}
$('#cp').onclick=copy;
$('#sh').onclick=function(){if(!url){say('Add your live link in SITE_URL first');return}
  if(navigator.share)navigator.share({title:'Nabia & Saad',text:text,url:url}).catch(function(e){if(e&&e.name!=='AbortError')copy()});else copy()};
})();
</script>
